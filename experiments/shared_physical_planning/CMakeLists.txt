cmake_minimum_required(VERSION 3.20)
project(ctdp_shared_physical_planning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

foreach(name IN ITEMS index_geometry physical_recursive compositional_pareto pareto_owned_labels)
    if(name STREQUAL "index_geometry")
        set(source "../../tests/space/test_physical_index_geometry.cpp")
    elseif(name STREQUAL "physical_recursive")
        set(source "../../tests/solver/test_physical_recursive.cpp")
    elseif(name STREQUAL "compositional_pareto")
        set(source "../../tests/solver/test_compositional_pareto.cpp")
    else()
        set(source "test_pareto_owned_labels.cpp")
    endif()
    add_executable(ctdp_${name} ${source})
    target_include_directories(ctdp_${name} PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
    if(MSVC)
        target_compile_options(ctdp_${name} PRIVATE /W4 /permissive- /O2)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(ctdp_${name} PRIVATE
            -Wall -Wextra -Wpedantic -Wconversion -O2 -fno-fast-math)
    endif()
endforeach()
enable_testing()
add_test(NAME ctdp_index_geometry COMMAND ctdp_index_geometry)
add_test(NAME ctdp_physical_recursive COMMAND ctdp_physical_recursive)
add_test(NAME ctdp_compositional_pareto COMMAND ctdp_compositional_pareto)

add_test(NAME ctdp_pareto_owned_labels COMMAND ctdp_pareto_owned_labels)
