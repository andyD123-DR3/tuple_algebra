cmake_minimum_required(VERSION 3.20)
project(ctdp_common_optimisation_session LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(ctdp_common_optimisation_session test_session.cpp)
target_include_directories(ctdp_common_optimisation_session PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
if(MSVC)
    target_compile_options(ctdp_common_optimisation_session PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(ctdp_common_optimisation_session PRIVATE
        -O2 -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
enable_testing()
add_test(NAME ctdp_common_optimisation_session COMMAND ctdp_common_optimisation_session)
